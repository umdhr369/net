use std::net::TcpStream;
use std::io::{Write};
use std::env;

fn main() {
    // 引数でサーバーIPを指定
    let args: Vec<String> = env::args().collect();
    if args.len() < 2 {
        eprintln!("使い方: client <server_ip>");
        return;
    }

    let server_ip = &args[1];
    let addr = format!("{}:5000", server_ip);

    println!("接続中: {}", addr);

    let mut stream = TcpStream::connect(&addr).expect("接続失敗");

    let msg = "こんにちは、サーバー！";
    stream.write_all(msg.as_bytes()).expect("送信失敗");

    println!("送信完了");
}
