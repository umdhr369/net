use std::io::{Read, Write};
use std::net::TcpStream;

fn main() -> std::io::Result<()> {
    // 【重要】"192.168.x.x" を、サーバー側Ubuntuの実際のIPアドレスに書き換えてください
    let server_ip = "192.168.x.x:8080"; 
    
    let mut stream = TcpStream::connect(server_ip)?;
    println!("サーバー({}) に接続しました。", server_ip);

    let msg = "Hello from another Ubuntu!";
    stream.write_all(msg.as_bytes())?;

    let mut buffer = [0; 512];
    let bytes_read = stream.read(&mut buffer)?;
    println!("サーバーからの応答: {}", String::from_utf8_lossy(&buffer[..bytes_read]));

    Ok(())
}
